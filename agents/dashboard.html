💻 Quick Start Example

from orchestrator import SalesOrchestrator # Initialize the orchestrator orchestrator = SalesOrchestrator() # Process a new lead lead = orchestrator.process_lead( phone="+1-555-0123", email="john.doe@example.com", name="John Doe", source="web_form" ) # Handle incoming call call = orchestrator.handle_incoming_call( phone="+1-555-0123", lead_id=lead["lead_id"] ) # Create sales deal deal = orchestrator.create_sales_deal( lead_id=lead["lead_id"], product_id="PROFESSIONAL", discount_percent=10 ) # Process payment payment = orchestrator.process_payment( lead_id=lead["lead_id"], deal_id=deal["deal_id"], amount=89.99, payment_method="credit_card" ) # Get system status status = orchestrator.get_system_status() print(f"Active leads: {status['statistics']['total_leads']}") print(f"Open deals: {status['statistics']['total_deals']}")
